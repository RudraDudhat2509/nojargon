# No-Jargon Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a Manifest V3 browser extension whose side panel gives a due-diligence verdict on any company page — what they actually do, a deterministic substance-vs-vapor score, decoded claims, and red flags.

**Architecture:** Fully client-side. A pure-TS `engine` computes a deterministic substance score from page text (never LLM-derived); a `lexicon` data module supplies buzzword→plain mappings; a content script extracts main copy with Readability and runs the engine; the side panel renders deterministic results first, then requests optional LLM prose from a background worker that selects between an on-device Nano adapter and a user-key Claude adapter.

**Tech Stack:** Manifest V3, TypeScript 5, Vite 5 + `@crxjs/vite-plugin`, `@mozilla/readability`, Vitest (+ `linkedom` for DOM in tests), vanilla TS/CSS panel.

## Global Constraints

- Node 20+, TypeScript 5.x, Manifest V3 only. Verbatim from design §8.
- No React in v1. Vanilla TS + CSS panel.
- `engine` and `lexicon` are pure: no DOM, no network, no `Date`/`Math.random`/locale-dependent calls (required for M3 determinism = 0 variance).
- Nothing deterministic (score, flags, decodes) may depend on the LLM (design §6 invariant).
- LLM layer order: on-device Nano → user's Claude key → omit prose. No hosted backend, ever.
- Claude model for `ClaudeAdapter`: `claude-sonnet-5`. Browser calls require header `anthropic-dangerous-direct-browser-access: true` and `anthropic-version: 2023-06-01`. Consult the `claude-api` skill before implementing that task.
- Commits: no AI co-authorship trailer.
- Metric gates come from `metrics-justification.md`: M1 ρ≥0.70, M2 ≥90%, M3 variance=0, M4 ≥80%.

```ts
import { DATA } from './data';
export interface LexEntry { term: string; plain: string; weight: number; empty: boolean; }
export const LEXICON: LexEntry[] = DATA;

// Longest terms first so multiword phrases win over their sub-words.
const SORTED = [...LEXICON].sort((a, b) => b.term.length - a.term.length);

export function findMatches(text: string): { entry: LexEntry; span: [number, number] }[] {
  const lower = text.toLowerCase();
  const taken: boolean[] = new Array(lower.length).fill(false);
  const out: { entry: LexEntry; span: [number, number] }[] = [];
  for (const entry of SORTED) {
    let from = 0;
    for (;;) {
      const i = lower.indexOf(entry.term, from);
      if (i === -1) break;
      const end = i + entry.term.length;
      const boundaryOk = (i === 0 || !/[a-z0-9]/.test(lower[i - 1]!)) &&
                         (end === lower.length || !/[a-z0-9]/.test(lower[end]!));
      const free = !taken.slice(i, end).some(Boolean);
      if (boundaryOk && free) { for (let k = i; k < end; k++) taken[k] = true; out.push({ entry, span: [i, end] }); }
      from = end;
    }
  }
  return out.sort((a, b) => a.span[0] - b.span[0]);
}
```

```ts
export function wordCount(text: string): number {
  return (text.match(/[A-Za-z][A-Za-z'-]*/g) ?? []).length;

// Each regex = one class of concrete signal. Count total matches.
const SIGNALS: RegExp[] = [
  /\b\d[\d,.]*\b/g,                         // numbers
  /[$₹€£]\s?\d/g,                           // currency
  /\b\d+\s?(ms|s|kb|mb|gb|tb|%|x)\b/gi,     // units
  /\b(19|20)\d{2}\b/g,                      // years
  /\b(REST|API|SDK|SLA|OAuth|SDKs?|webhook|integration|integrations)\b/gi, // tech specifics
  /\b[A-Z][a-z]+ (?:Inc|Ltd|LLC|Corp|GmbH|Co)\b/g, // named orgs
];
export function concreteCount(text: string): number {
  return SIGNALS.reduce((n, re) => n + (text.match(re)?.length ?? 0), 0);
}
```

```ts
import { findMatches } from '../lexicon';
import { wordCount, concreteCount } from './signals';
import { deriveRedFlags } from './flags';
import type { ScoreResult, DetectedClaim } from './types';
export * from './types';

export function score(text: string): ScoreResult {
  const words = wordCount(text);
  const matches = findMatches(text);
  const fluffRaw = matches.reduce((s, m) => s + m.entry.weight, 0);
  const concreteRaw = concreteCount(text);
  const per1k = (n: number) => (words === 0 ? 0 : (n / words) * 1000);
  const fluffPer1k = per1k(fluffRaw);
  const concretePer1k = per1k(concreteRaw);
  const denom = fluffPer1k + concretePer1k;
  const substancePct = denom === 0 ? null : Math.round((100 * concretePer1k) / denom);

  const claims: DetectedClaim[] = matches.map(m => ({
    text: text.slice(m.span[0], m.span[1]),
    span: m.span, plain: m.entry.plain, empty: m.entry.empty,
  }));
  const redFlags = deriveRedFlags(text, { fluffPer1k, concretePer1k, words });
  return { substancePct, fluffPer1k, concretePer1k, words, claims, redFlags };
}
```

```ts
import type { RedFlag } from './types';
const DENSITY_THRESHOLD = 20; // fluff points per 1k words; calibrated with M1 golden set
export function deriveRedFlags(text: string, m: { fluffPer1k: number; concretePer1k: number; words: number }): RedFlag[] {
  const flags: RedFlag[] = [];
  if (m.fluffPer1k >= DENSITY_THRESHOLD)
    flags.push({ id: 'high_buzzword_density', message: 'Heavy on buzzwords relative to concrete detail.' });
  if (m.concretePer1k === 0)
    flags.push({ id: 'no_numbers_or_specs', message: 'No numbers, prices, or technical specifics found.' });
  if (!/\b[A-Z][a-z]+ (?:Inc|Ltd|LLC|Corp|GmbH|Co)\b/.test(text) && !/\bcustomers?\b/i.test(text))
    flags.push({ id: 'no_named_customers', message: 'No named customers or case studies.' });
  return flags;
}
```

```ts
import { Readability } from '@mozilla/readability';
import { score } from '../engine';
import type { ExtractResponse } from '../shared/messages';

export function extractMain(doc: Document): string | null {
  const clone = doc.cloneNode(true) as Document;
  const parsed = new Readability(clone).parse();
  const text = (parsed?.textContent ?? '').trim();
  return text.length >= 40 ? text : null;
}

// Chrome runtime wiring (not exercised in unit tests)
if (typeof chrome !== 'undefined' && chrome.runtime?.onMessage) {
  chrome.runtime.onMessage.addListener((msg, _sender, sendResponse) => {
    if (msg?.type !== 'extract') return;
```

```ts
import type { DetectedClaim, Label } from '../engine/types';
export interface LlmAdapter {
  name: string;
  isAvailable(): Promise<boolean>;
  enrich(mainText: string, claims: DetectedClaim[]): Promise<{ whatTheyDo: string; claimLabels: Record<number, Label> }>;
}
export async function selectAdapter(candidates: LlmAdapter[]): Promise<LlmAdapter | null> {
  for (const a of candidates) { if (await a.isAvailable()) return a; }
  return null;
}
export const PROMPT = (mainText: string) =>
```

```ts
export function spearman(xs: number[], ys: number[]): number {
  const rank = (a: number[]) => {
    const idx = a.map((v, i) => [v, i] as const).sort((p, q) => p[0] - q[0]);
    const r = new Array(a.length);
    idx.forEach(([, i], k) => { r[i] = k + 1; });
    return r;
  };
  const rx = rank(xs), ry = rank(ys), n = xs.length;
  const d2 = rx.reduce((s, v, i) => s + (v - ry[i]) ** 2, 0);
  return 1 - (6 * d2) / (n * (n * n - 1));
}
```
